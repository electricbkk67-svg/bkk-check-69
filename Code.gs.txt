const SPREADSHEET_ID = '1mDYlCX3TCY2mmGAytl6S7udFgt3BV5AWz7tm5KSgh_U'; // ใส่ Spreadsheet ID หรือวางลิงก์ Google Sheets ทั้งหมดที่นี่
const FOLDER_ID = '1TH0qnjf4DjDNuipY3gxrnkqcW53kMP19';      // ใส่ Folder ID หรือลิงก์โฟลเดอร์ Google Drive สำหรับเก็บรูป (ถ้าไม่ใส่จะสร้างให้อัตโนมัติ)

function extractSpreadsheetId(input) {
  if (!input) return '';
  const str = String(input).trim();
  const match = str.match(/\/spreadsheets\/d\/([a-zA-Z0-9-_]+)/);
  if (match && match[1]) return match[1];
  const match2 = str.match(/\/d\/([a-zA-Z0-9-_]+)/);
  if (match2 && match2[1]) return match2[1];
  return str;
}

function extractFolderId(input) {
  if (!input) return '';
  const str = String(input).trim();
  const match = str.match(/\/folders\/([a-zA-Z0-9-_]+)/);
  if (match && match[1]) return match[1];
  return str;
}

function getSpreadsheet() {
  let ss = null;
  const rawId = (typeof SPREADSHEET_ID !== 'undefined') ? SPREADSHEET_ID : '';
  const cleanId = extractSpreadsheetId(rawId);

  // 1. ลองเปิดจาก SPREADSHEET_ID (รองรับทั้ง ID และ Full URL ลิงก์ชีต)
  if (cleanId && cleanId !== '') {
    try {
      ss = SpreadsheetApp.openById(cleanId);
    } catch (err) {
      Logger.log('openById error: ' + err.toString());
    }
  }

  // 2. ถ้ายังไม่ได้ ให้ลองดึง Active Spreadsheet (กรณีเปิด Script จากเมนู ส่วนขยาย > Apps Script)
  if (!ss) {
    try {
      ss = SpreadsheetApp.getActiveSpreadsheet();
    } catch (err) {
      Logger.log('getActiveSpreadsheet error: ' + err.toString());
    }
  }

  // 3. Fallback อัตโนมัติ: ค้นหาไฟล์ชีตชื่อ "ตรวจสอบพัสดุ 2569" จาก Google Drive
  if (!ss) {
    try {
      const candidates = ['ตรวจสอบพัสดุ 2569', 'ตรวจสอบพัสดุ 2569 (ส่งกองคลัง)'];
      for (let i = 0; i < candidates.length; i++) {
        const files = DriveApp.getFilesByName(candidates[i]);
        while (files.hasNext()) {
          const file = files.next();
          try {
            ss = SpreadsheetApp.openById(file.getId());
            if (ss) break;
          } catch (e) {}
        }
        if (ss) break;
      }
    } catch (err) {
      Logger.log('Auto search spreadsheet error: ' + err.toString());
    }
  }

  // 4. หากยังไม่พบ แจ้งเตือนข้อความภาษาไทยอย่างชัดเจน
  if (!ss) {
    throw new Error('ไม่พบไฟล์ Google Sheets! กรุณานำลิงก์ Google Sheets (หรือ ID) ไปวางที่บรรทัด SPREADSHEET_ID ในไฟล์ Code.gs แล้ว Deploy เวอร์ชันใหม่ครับ');
  }

  return ss;
}

function getUploadFolder() {
  if (FOLDER_ID && FOLDER_ID.trim() !== '') {
    try {
      const cleanFolderId = extractFolderId(FOLDER_ID);
      return DriveApp.getFolderById(cleanFolderId);
    } catch (e) {}
  }
  const folders = DriveApp.getFoldersByName('รูปภาพตรวจสอบพัสดุ_2569');
  if (folders.hasNext()) {
    return folders.next();
  }
  const newFolder = DriveApp.createFolder('รูปภาพตรวจสอบพัสดุ_2569');
  newFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return newFolder;
}

function getThaiTimestamp() {
  const d = new Date();
  const bangkokTime = new Date(d.getTime() + (7 * 60 + d.getTimezoneOffset()) * 60000);
  const thaiMonths = ['ม.ค.', 'ก.พ.', 'มี.ค.', 'เม.ย.', 'พ.ค.', 'มิ.ย.', 'ก.ค.', 'ส.ค.', 'ก.ย.', 'ต.ค.', 'พ.ย.', 'ธ.ค.'];
  const day = bangkokTime.getDate();
  const month = thaiMonths[bangkokTime.getMonth()];
  const year = bangkokTime.getFullYear() + 543;
  const hours = String(bangkokTime.getHours()).padStart(2, '0');
  const mins = String(bangkokTime.getMinutes()).padStart(2, '0');
  return `${day} ${month} ${year} ${hours}:${mins} น.`;
}

function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : '';

  if (action === 'getAll') {
    const result = getAllInspectionData();
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (action === 'ping') {
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: 'เชื่อมต่อ Google Sheets สำเร็จ!'
    })).setMimeType(ContentService.MimeType.JSON);
  }

  // กรณีรันผ่าน Google Apps Script HTML Service โดยตรง
  try {
    return HtmlService.createTemplateFromFile('index')
      .evaluate()
      .setTitle('ระบบตรวจสอบพัสดุ 2569 - กองช่าง อบต.บางขุนกอง')
      .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      success: true,
      message: 'API ระบบตรวจสอบพัสดุพร้อมใช้งาน'
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    if (action === 'saveInspection') {
      const res = saveInspectionRecord(payload);
      return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'saveUnlisted') {
      const res = saveUnlistedRecord(payload);
      return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'deleteUnlisted') {
      const res = deleteUnlistedRecord(payload);
      return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'clearInspection') {
      const res = clearInspectionRecord(payload);
      return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (action === 'syncBatch') {
      const res = syncBatchRecords(payload);
      return ContentService.createTextOutput(JSON.stringify(res))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      message: 'ไม่รู้จักคำสั่ง: ' + action
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      message: 'เกิดข้อผิดพลาด: ' + err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function uploadBase64Image(imageFile) {
  if (!imageFile || !imageFile.data || imageFile.data.indexOf(',') === -1) return '';
  try {
    const folder = getUploadFolder();
    const contentType = imageFile.data.substring(5, imageFile.data.indexOf(';'));
    const bytes = Utilities.base64Decode(imageFile.data.substr(imageFile.data.indexOf(',') + 1));
    const filename = imageFile.name || ('asset_' + Date.now() + '.jpg');
    const blob = Utilities.newBlob(bytes, contentType, filename);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return 'https://lh3.googleusercontent.com/d/' + file.getId() + '=s0';
  } catch (err) {
    Logger.log('Upload image error: ' + err.toString());
    return '';
  }
}

/**
 * บันทึกผลการตรวจสอบลงในชีตต้นฉบับ ("ช่าง" หรือ "ทีดินและสิ่งปลูกสร้าง")
 * และบันทึกลงชีต "InspectionLog" เพื่อซิงค์ข้อมูลได้อย่างรวดเร็ว
 */
function saveInspectionRecord(data) {
  const ss = getSpreadsheet();
  const finalUrls = [];
  if (data.existingImageUrl) {
    String(data.existingImageUrl).split(',').map(s => s.trim()).filter(Boolean).forEach(u => finalUrls.push(u));
  }
  if (Array.isArray(data.imageFiles) && data.imageFiles.length > 0) {
    data.imageFiles.forEach(imgFile => {
      const uploaded = uploadBase64Image(imgFile);
      if (uploaded) finalUrls.push(uploaded);
    });
  } else if (data.imageFile && data.imageFile.data) {
    const uploaded = uploadBase64Image(data.imageFile);
    if (uploaded) finalUrls.push(uploaded);
  }
  const imageUrl = finalUrls.join(',');

  const timestamp = getThaiTimestamp();
  const statusMap = {
    'usable': 0,      // มี - ใช้ได้
    'damaged': 1,     // มี - ชำรุด
    'deteriorated': 2,// มี - เสื่อมสภาพ
    'unneeded': 3,    // มี - ไม่จำเป็นใช้
    'missing': 4,     // ไม่พบ
    'lost': 5         // สูญไป
  };

  // 1. อัปเดตลงชีตจริง (ค้นหาแถวจาก รหัสพัสดุ / รหัสเต็ม / ลำดับ เพื่อป้องกันปัญหาเมื่อติด Filter หรือ Sort อยู่)
  const targetSheetName = data.sheet || 'ช่าง';
  let ws = ss.getSheetByName(targetSheetName);
  if (!ws && (targetSheetName.indexOf('ดิน') !== -1)) {
    ws = ss.getSheetByName('ที่ดินและสิ่งปลูกสร้าง') || ss.getSheetByName('ทีดินและสิ่งปลูกสร้าง');
  }
  if (ws) {
    const isSheet1 = (targetSheetName === 'ช่าง');
    const startStatusCol = isSheet1 ? 11 : 12; // ช่าง: K(11)-P(16), ที่ดิน: L(12)-Q(17)
    const locCol = isSheet1 ? 17 : 18;
    const noteCol = isSheet1 ? 19 : 20;
    const imgCol = isSheet1 ? 20 : 21;
    const timeCol = isSheet1 ? 21 : 22;

    const lastRow = ws.getLastRow();
    let actualRow = -1;

    if (lastRow >= 6) {
      // ดึงข้อมูลคอลัมน์ A ถึง J มาตรวจสอบหาแถวที่ถูกต้อง
      // Sheet ช่าง: Col A(0)=ลำดับ, Col D(3)=รหัสเต็ม, Col G(6)=เลขที่หรือรหัส, Col H(7)=ชื่อพัสดุ
      // Sheet ที่ดิน: Col E(4)=ลำดับ, Col D(3)=รหัสเต็ม, Col H(7)=เลขที่หรือรหัส, Col I(8)=ชื่อพัสดุ
      const maxColCheck = 10;
      const values = ws.getRange(1, 1, lastRow, maxColCheck).getDisplayValues();

      const searchCode = String(data.code || '').trim();
      const searchFullCode = String(data.fullCode || '').trim();
      const searchSeq = String(data.seq || '').trim();
      const searchName = String(data.name || '').trim();

      // ค้นหาแถวที่ตรงกันจริง
      for (let r = 5; r < values.length; r++) {
        const row = values[r];
        const rowCode = isSheet1 ? String(row[6] || '').trim() : String(row[7] || '').trim();
        const rowFullCode = String(row[3] || '').trim();
        const rowSeq = isSheet1 ? String(row[0] || '').trim() : String(row[4] || '').trim();
        const rowName = isSheet1 ? String(row[7] || '').trim() : String(row[8] || '').trim();

        // ตรวจสอบความตรงกัน: เลขรหัสตรง หรือ (ลำดับตรงและชื่อตรง)
        if (searchCode && rowCode && rowCode === searchCode) {
          actualRow = r + 1;
          break;
        } else if (searchFullCode && rowFullCode && rowFullCode === searchFullCode) {
          actualRow = r + 1;
          break;
        } else if (searchSeq && rowSeq && searchSeq === rowSeq && searchName && rowName && searchName === rowName) {
          actualRow = r + 1;
          break;
        }
      }
    }

    // หากค้นหาในชีตไม่เจอ ให้ใช้ rowIndex เดิมเป็นตัวสำรอง
    if (actualRow === -1 && data.rowIndex) {
      actualRow = parseInt(data.rowIndex, 10);
    }

    if (actualRow > 0 && actualRow <= lastRow) {
      const statusValues = [['', '', '', '', '', '']];
      if (data.status && statusMap[data.status] !== undefined) {
        statusValues[0][statusMap[data.status]] = '/';
      }
      ws.getRange(actualRow, startStatusCol, 1, 6).setValues(statusValues);
      if (data.location !== undefined) ws.getRange(actualRow, locCol).setValue(data.location);
      if (data.note !== undefined) ws.getRange(actualRow, noteCol).setValue(data.note);
      if (imageUrl) ws.getRange(actualRow, imgCol).setValue(imageUrl);
      ws.getRange(actualRow, timeCol).setValue(timestamp);
    }
  }

  // 2. บันทึกลงชีตฐานข้อมูลสถานะ "InspectionLog"
  let logSheet = ss.getSheetByName('InspectionLog');
  if (!logSheet) {
    logSheet = ss.insertSheet('InspectionLog');
    logSheet.appendRow(['id', 'sheet', 'rowIndex', 'code', 'name', 'status', 'location', 'note', 'imageUrl', 'checkedAt']);
  }

  const lastRow = logSheet.getLastRow();
  let foundRow = -1;
  if (lastRow > 1) {
    const ids = logSheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) === String(data.id)) {
        foundRow = i + 2;
        break;
      }
    }
  }

  const rowValues = [[
    data.id,
    data.sheet || 'ช่าง',
    data.rowIndex || '',
    data.code || '',
    data.name || '',
    data.status || '',
    data.location || '',
    data.note || '',
    imageUrl,
    timestamp
  ]];

  if (foundRow !== -1) {
    logSheet.getRange(foundRow, 1, 1, 10).setValues(rowValues);
  } else {
    logSheet.appendRow(rowValues[0]);
  }

  return {
    success: true,
    message: 'บันทึกผลการตรวจสอบเรียบร้อย',
    imageUrl: imageUrl,
    checkedAt: timestamp
  };
}

/**
 * ล้างผลการตรวจสอบของรายการนี้ในชีตต้นฉบับ ("ช่าง" หรือ "ทีดินและสิ่งปลูกสร้าง")
 * และลบออกจากชีต "InspectionLog"
 */
function clearInspectionRecord(data) {
  const ss = getSpreadsheet();
  const targetSheetName = data.sheet || 'ช่าง';
  let ws = ss.getSheetByName(targetSheetName);
  if (!ws && (targetSheetName.indexOf('ดิน') !== -1)) {
    ws = ss.getSheetByName('ที่ดินและสิ่งปลูกสร้าง') || ss.getSheetByName('ทีดินและสิ่งปลูกสร้าง');
  }

  if (ws) {
    const isSheet1 = (targetSheetName === 'ช่าง');
    const startStatusCol = isSheet1 ? 11 : 12; // ช่าง: K(11)-P(16), ที่ดิน: L(12)-Q(17)
    const locCol = isSheet1 ? 17 : 18;
    const noteCol = isSheet1 ? 19 : 20;
    const imgCol = isSheet1 ? 20 : 21;
    const timeCol = isSheet1 ? 21 : 22;

    const lastRow = ws.getLastRow();
    let actualRow = -1;

    if (lastRow >= 6) {
      const maxColCheck = 10;
      const values = ws.getRange(1, 1, lastRow, maxColCheck).getDisplayValues();

      const searchCode = String(data.code || '').trim();
      const searchFullCode = String(data.fullCode || '').trim();
      const searchSeq = String(data.seq || '').trim();
      const searchName = String(data.name || '').trim();

      for (let r = 5; r < values.length; r++) {
        const row = values[r];
        const rowCode = isSheet1 ? String(row[6] || '').trim() : String(row[7] || '').trim();
        const rowFullCode = String(row[3] || '').trim();
        const rowSeq = isSheet1 ? String(row[0] || '').trim() : String(row[4] || '').trim();
        const rowName = isSheet1 ? String(row[7] || '').trim() : String(row[8] || '').trim();

        if (searchCode && rowCode && rowCode === searchCode) {
          actualRow = r + 1;
          break;
        } else if (searchFullCode && rowFullCode && rowFullCode === searchFullCode) {
          actualRow = r + 1;
          break;
        } else if (searchSeq && rowSeq && searchSeq === rowSeq && searchName && rowName && searchName === rowName) {
          actualRow = r + 1;
          break;
        }
      }
    }

    if (actualRow === -1 && data.rowIndex) {
      actualRow = parseInt(data.rowIndex, 10);
    }

    if (actualRow > 0 && actualRow <= lastRow) {
      // 1. ล้างผลการตรวจทั้ง 6 ช่องให้ว่างเปล่า
      ws.getRange(actualRow, startStatusCol, 1, 6).setValues([['', '', '', '', '', '']]);

      // 2. คืนค่าช่องสถานที่
      const origLoc = (data.originalLocation !== undefined) ? data.originalLocation : '';
      ws.getRange(actualRow, locCol).setValue(origLoc);

      // 3. คืนค่าช่องหมายเหตุเดิม (เช่น 'รับเพิ่ม 2569' หรือล้างออก)
      const origNote = (data.originalNote !== undefined) ? data.originalNote : '';
      ws.getRange(actualRow, noteCol).setValue(origNote);

      // 4. ล้างรูปภาพและเวลาบันทึก
      ws.getRange(actualRow, imgCol).setValue('');
      ws.getRange(actualRow, timeCol).setValue('');
    }
  }

  // ลบแถวออกจาก InspectionLog เพื่อไม่ให้ sync กลับมา
  const logSheet = ss.getSheetByName('InspectionLog');
  if (logSheet && logSheet.getLastRow() > 1) {
    const ids = logSheet.getRange(2, 1, logSheet.getLastRow() - 1, 1).getValues();
    for (let i = ids.length - 1; i >= 0; i--) {
      if (String(ids[i][0]) === String(data.id)) {
        logSheet.deleteRow(i + 2);
      }
    }
  }

  return {
    success: true,
    message: 'ล้างผลการตรวจสอบในชีตเรียบร้อย'
  };
}

/**
 * บันทึกพัสดุนอกบัญชี / ค้นไม่พบในระบบ ลงชีต "พัสดุนอกบัญชี"
 */
function saveUnlistedRecord(data) {
  const ss = getSpreadsheet();
  let ulSheet = ss.getSheetByName('พัสดุนอกบัญชี');
  if (!ulSheet) {
    ulSheet = ss.insertSheet('พัสดุนอกบัญชี');
    ulSheet.appendRow(['id', 'code', 'name', 'category', 'status', 'location', 'note', 'imageUrl', 'checkedAt']);
  }

  const finalUrls = [];
  if (data.existingImageUrl) {
    String(data.existingImageUrl).split(',').map(s => s.trim()).filter(Boolean).forEach(u => finalUrls.push(u));
  }
  if (Array.isArray(data.imageFiles) && data.imageFiles.length > 0) {
    data.imageFiles.forEach(imgFile => {
      const uploaded = uploadBase64Image(imgFile);
      if (uploaded) finalUrls.push(uploaded);
    });
  } else if (data.imageFile && data.imageFile.data) {
    const uploaded = uploadBase64Image(data.imageFile);
    if (uploaded) finalUrls.push(uploaded);
  }
  const imageUrl = finalUrls.join(',');

  const timestamp = data.checkedAt || getThaiTimestamp();
  const id = data.id || ('UL_' + Date.now());

  const lastRow = ulSheet.getLastRow();
  let foundRow = -1;
  if (lastRow > 1) {
    const ids = ulSheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) === String(id)) {
        foundRow = i + 2;
        break;
      }
    }
  }

  const rowData = [
    id,
    data.code || '',
    data.name || '',
    data.category || 'ครุภัณฑ์นอกบัญชี',
    data.status || 'usable',
    data.location || '',
    data.note || '',
    imageUrl,
    timestamp
  ];

  if (foundRow !== -1) {
    ulSheet.getRange(foundRow, 1, 1, 9).setValues([rowData]);
  } else {
    ulSheet.appendRow(rowData);
  }

  return {
    success: true,
    message: 'บันทึกพัสดุนอกบัญชีเรียบร้อย',
    id: id,
    imageUrl: imageUrl,
    checkedAt: timestamp
  };
}

function deleteUnlistedRecord(data) {
  const ss = getSpreadsheet();
  const ulSheet = ss.getSheetByName('พัสดุนอกบัญชี');
  if (!ulSheet) return { success: true };

  const lastRow = ulSheet.getLastRow();
  if (lastRow > 1) {
    const ids = ulSheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = ids.length - 1; i >= 0; i--) {
      if (String(ids[i][0]) === String(data.id)) {
        ulSheet.deleteRow(i + 2);
      }
    }
  }
  return { success: true, message: 'ลบรายการนอกบัญชีเรียบร้อย' };
}

          note: r[7],

        };
      }
    });
  }

  const ulSheet = ss.getSheetByName('พัสดุนอกบัญชี');

}

function syncBatchRecords(payload) {
  const ss = getSpreadsheet();
  const items = payload.items || [];
  const unlisted = payload.unlisted || [];
  const statusMap = {
    'usable': 0,
    'damaged': 1,
    'deteriorated': 2,
    'unneeded': 3,
    'missing': 4,
    'lost': 5
  };

  // แยกรายการตามชีต
  const sheet1Items = [];
  const sheet2Items = [];
  items.forEach(it => {
    const sName = String(it.sheet || '').trim();
    if (sName.indexOf('ดิน') !== -1) {
      sheet2Items.push(it);
    } else {
      sheet1Items.push(it);
    }
  });

  // 1. อัปเดตชีต 1 ("ช่าง") แบบ Batch ครั้งเดียว
  if (sheet1Items.length > 0) {
    const ws1 = ss.getSheetByName('ช่าง');
    if (ws1 && ws1.getLastRow() >= 6) {
      const lastRow = ws1.getLastRow();
      const numDataRows = lastRow - 5; // แถว 6 ถึง lastRow
      const lookupValues = ws1.getRange(6, 1, numDataRows, 10).getDisplayValues();

      // ป้องกันเรื่องขนาดคอลัมน์เกินขอบเขตตารางของชีต (ช่างมีปกติถึง Col S = 19)
      const maxCols1 = ws1.getMaxColumns();
      const numWriteCols = (maxCols1 >= 21) ? 11 : 9; // 9 คอลัมน์คือ K(11) ถึง S(19)
      const targetValues = ws1.getRange(6, 11, numDataRows, numWriteCols).getValues();

      const rowMapByCode = {};
      const rowMapByFullCode = {};
      const rowMapBySeqName = {};

      for (let i = 0; i < lookupValues.length; i++) {
        const row = lookupValues[i];
        const rowSeq = String(row[0] || '').trim();
        const rowFullCode = String(row[3] || '').trim();
        const rowCode = String(row[6] || '').trim();
        const rowName = String(row[7] || '').trim();

        if (rowCode) rowMapByCode[rowCode] = i;
        if (rowFullCode) rowMapByFullCode[rowFullCode] = i;
        if (rowSeq && rowName) rowMapBySeqName[rowSeq + '|||' + rowName] = i;
      }

      sheet1Items.forEach(it => {
        const sCode = String(it.code || '').trim();
        const sFull = String(it.fullCode || '').trim();
        const sSeq = String(it.seq || '').trim();
        const sName = String(it.name || '').trim();

        let rowIdx = -1;
        if (sCode && rowMapByCode[sCode] !== undefined) rowIdx = rowMapByCode[sCode];
        else if (sFull && rowMapByFullCode[sFull] !== undefined) rowIdx = rowMapByFullCode[sFull];
        else if (sSeq && sName && rowMapBySeqName[sSeq + '|||' + sName] !== undefined) rowIdx = rowMapBySeqName[sSeq + '|||' + sName];
        else if (it.rowIndex && it.rowIndex >= 6 && (it.rowIndex - 6) < numDataRows) rowIdx = it.rowIndex - 6;

        if (rowIdx >= 0) {
          for (let c = 0; c < 6; c++) targetValues[rowIdx][c] = '';
          if (it.status && statusMap[it.status] !== undefined) {
            targetValues[rowIdx][statusMap[it.status]] = '/';
          }
          if (it.location !== undefined && it.location !== '') targetValues[rowIdx][6] = it.location; // Col Q
          // targetValues[rowIdx][7] คือ Col R (เจ้าของงบประมาณ) คงเดิมไว้
          if (it.note !== undefined && it.note !== '') targetValues[rowIdx][8] = it.note; // Col S
          if (numWriteCols >= 11) {
            if (it.imageUrl) targetValues[rowIdx][9] = it.imageUrl; // Col T
            targetValues[rowIdx][10] = it.checkedAt || getThaiTimestamp(); // Col U
          }
        }
      });

      ws1.getRange(6, 11, numDataRows, numWriteCols).setValues(targetValues);
    }
  }

  // 2. อัปเดตชีต 2 ("ทีดินและสิ่งปลูกสร้าง") แบบ Batch ครั้งเดียว
  if (sheet2Items.length > 0) {
    let ws2 = ss.getSheetByName('ทีดินและสิ่งปลูกสร้าง') || ss.getSheetByName('ที่ดินและสิ่งปลูกสร้าง');
    if (ws2 && ws2.getLastRow() >= 7) {
      const lastRow = ws2.getLastRow();
      const numDataRows = lastRow - 6; // แถว 7 ถึง lastRow
      const lookupValues = ws2.getRange(7, 1, numDataRows, 11).getDisplayValues();

      // ป้องกันเรื่องขนาดคอลัมน์เกินขอบเขตตารางของชีต (ที่ดินมีปกติถึง Col T = 20)
      const maxCols2 = ws2.getMaxColumns();
      const numWriteCols = (maxCols2 >= 22) ? 11 : 9; // 9 คอลัมน์คือ L(12) ถึง T(20)
      const targetValues = ws2.getRange(7, 12, numDataRows, numWriteCols).getValues();

      const rowMapByCode = {};
      const rowMapByFullCode = {};
      const rowMapBySeqName = {};

      for (let i = 0; i < lookupValues.length; i++) {
        const row = lookupValues[i];
        const rowFullCode = String(row[3] || '').trim();
        const rowSeq = String(row[4] || '').trim();
        const rowCode = String(row[7] || '').trim();
        const rowName = String(row[8] || '').trim();

        if (rowCode) rowMapByCode[rowCode] = i;
        if (rowFullCode) rowMapByFullCode[rowFullCode] = i;
        if (rowSeq && rowName) rowMapBySeqName[rowSeq + '|||' + rowName] = i;
      }

      sheet2Items.forEach(it => {
        const sCode = String(it.code || '').trim();
        const sFull = String(it.fullCode || '').trim();
        const sSeq = String(it.seq || '').trim();
        const sName = String(it.name || '').trim();

        let rowIdx = -1;
        if (sCode && rowMapByCode[sCode] !== undefined) rowIdx = rowMapByCode[sCode];
        else if (sFull && rowMapByFullCode[sFull] !== undefined) rowIdx = rowMapByFullCode[sFull];
        else if (sSeq && sName && rowMapBySeqName[sSeq + '|||' + sName] !== undefined) rowIdx = rowMapBySeqName[sSeq + '|||' + sName];
        else if (it.rowIndex && it.rowIndex >= 7 && (it.rowIndex - 7) < numDataRows) rowIdx = it.rowIndex - 7;

        if (rowIdx >= 0) {
          for (let c = 0; c < 6; c++) targetValues[rowIdx][c] = '';
          if (it.status && statusMap[it.status] !== undefined) {
            targetValues[rowIdx][statusMap[it.status]] = '/';
          }
          if (it.location !== undefined && it.location !== '') targetValues[rowIdx][6] = it.location; // Col R
          // targetValues[rowIdx][7] คือ Col S (เจ้าของงบประมาณ) คงเดิมไว้
          if (it.note !== undefined && it.note !== '') targetValues[rowIdx][8] = it.note; // Col T
          if (numWriteCols >= 11) {
            if (it.imageUrl) targetValues[rowIdx][9] = it.imageUrl; // Col U
            targetValues[rowIdx][10] = it.checkedAt || getThaiTimestamp(); // Col V
          }
        }
      });

      ws2.getRange(7, 12, numDataRows, numWriteCols).setValues(targetValues);
    }
  }

  // 3. อัปเดตชีต "InspectionLog" แบบ Batch ครบทุกรายการแน่นอน 100%
  const LOG_COLS = 10;
  let logSheet = ss.getSheetByName('InspectionLog');
  if (!logSheet) {
    logSheet = ss.insertSheet('InspectionLog');
  }
  // ปรับขนาดคอลัมน์ของชีต InspectionLog ให้มีอย่างน้อย 10 คอลัมน์
  if (logSheet.getMaxColumns() < LOG_COLS) {
    logSheet.insertColumnsAfter(logSheet.getMaxColumns(), LOG_COLS - logSheet.getMaxColumns());
  }

  const lastLogRow = logSheet.getLastRow();
  let existingLogData = [];
  const logMapById = {};

  if (lastLogRow > 0) {
    // อ่านเฉพาะ 10 คอลัมน์แรก เพื่อให้ขนาดอาร์เรย์เป็น 10 คอลัมน์เสมอ
    existingLogData = logSheet.getRange(1, 1, lastLogRow, LOG_COLS).getValues();
    for (let r = 1; r < existingLogData.length; r++) {
      const logId = String(existingLogData[r][0] || '').trim();
      if (logId) logMapById[logId] = r;
    }
  } else {
    existingLogData.push(['id', 'sheet', 'rowIndex', 'code', 'name', 'status', 'location', 'note', 'imageUrl', 'checkedAt']);
  }

  const newLogRows = [];
  items.forEach(it => {
    const itId = String(it.id || '').trim();
    const rowData = [
      it.id || '',
      it.sheet || 'ช่าง',
      it.rowIndex || '',
      it.code || '',
      it.name || '',
      it.status || '',
      it.location || '',
      it.note || '',
      it.imageUrl || '',
      it.checkedAt || getThaiTimestamp()
    ];

    if (itId && logMapById[itId] !== undefined) {
      existingLogData[logMapById[itId]] = rowData;
    } else {
      newLogRows.push(rowData);
      logMapById[itId] = existingLogData.length + newLogRows.length - 1;
    }
  });

  // รวมข้อมูลทั้งหมดเขียนลงชีตใน 1 คำสั่งเดียว (ไม่มีทางหลุดหรือขาดหาย)
  const allLogData = existingLogData.concat(newLogRows);
  if (allLogData.length > 0) {
    logSheet.getRange(1, 1, allLogData.length, LOG_COLS).setValues(allLogData);
  }

  // 4. พัสดุนอกบัญชี
  if (unlisted.length > 0) {
    unlisted.forEach(ul => {
      saveUnlistedRecord(ul);
    });
  }

  return {
    success: true,
    message: `ซิงค์ข้อมูลสำเร็จ (${items.length} รายการ: กองช่าง ${sheet1Items.length}, ที่ดิน/สิ่งปลูกสร้าง ${sheet2Items.length})`
  };
}
